void Clear_Buffer(void)//清空缓存
{
	printf("\r\nResponse: %s\r\n", buf_uart2.buf);
	strx=strstr((const char*)buf_uart2.buf,(const char*)"+QIURC");//返回+QIURC:，表明接收到TCP服务器发回的数据	
    delay_ms(300);
    buf_uart2.index=0;
    memset(buf_uart2.buf,0,BUFLEN);
	IWDG_Feed();//喂狗
}

/*****************************************************
初始化模块 和单片机连接，获取卡号和信号质量
*****************************************************/
void CSTX_4G_Init(void)
{
	u8 retry_count = 0;//定位次数
	//打印初始化信息
	printf("\r\n start init EC200X\r\n");
	//发第一个命令ATE1
    Uart2_SendStr("ATE1\r\n"); 
    delay_ms(300);
	printf("\r\nResponse: %s\r\n", buf_uart2.buf);  // 明确格式化输出
    strx=strstr((const char*)buf_uart2.buf,(const char*)"OK");//返回OK
    Clear_Buffer();	
    while(strx==NULL)
    {
		printf("单片机正在连接模块......\r\n");
        Clear_Buffer();	
        Uart2_SendStr("ATE1\r\n"); 
        delay_ms(300);
        strx=strstr((const char*)buf_uart2.buf,(const char*)"OK");//返回OK
    }
	printf("****单片机和模块连接成功*****\r\n");
	Uart2_SendStr("ATI\r\n");//获取模块的版本
	delay_ms(300);
	Clear_Buffer();	
		
    Uart2_SendStr("AT+CIMI\r\n");//获取卡号，类似是否存在卡的意思，比较重要。
    delay_ms(300);
    strx=strstr((const char*)buf_uart2.buf,(const char*)"460");//返460，表明识别到卡了
    while(strx==NULL)
    {
        Clear_Buffer();	
        Uart2_SendStr("AT+CIMI\r\n");//获取卡号，类似是否存在卡的意思，比较重要。
        delay_ms(300);
        strx=strstr((const char*)buf_uart2.buf,(const char*)"460");//返回OK,说明卡是存在的
    } 
	printf("我的卡号是 : %s \r\n",buf_uart2.buf+8);
	Clear_Buffer();	
		
	Uart2_SendStr("AT+CGATT?\r\n");//查询激活状态
	delay_ms(300);
	strx=strstr((const char*)buf_uart2.buf,(const char*)"+CGATT: 1");//返1
	Clear_Buffer();	
	while(strx==NULL)
	{
		Clear_Buffer();	
		Uart2_SendStr("AT+CGATT?\r\n");//获取激活状态
		delay_ms(300);
		strx=strstr((const char*)buf_uart2.buf,(const char*)"+CGATT: 1");//返回1,表明注网成功
	}
	Clear_Buffer();	
	Uart2_SendStr("AT+CSQ\r\n");//查看获取CSQ值
	delay_ms(300);
    strx=strstr((const char*)buf_uart2.buf,(const char*)"+CSQ:");//返回CSQ
	if(strx)
	{
		printf("信号质量是:%s 注意：信号最大值是31 \r\n",buf_uart2.buf+14);    
		Clear_Buffer();
		
		Uart2_SendStr("AT+QGPS=1\r\n");
		delay_ms(300);
		strx = strstr((const char*)buf_uart2.buf, (const char*)"+CMS ERROR: 504");
		while(strx == NULL)
		{
			Clear_Buffer();
			Uart2_SendStr("AT+QGPS=1\r\n");
			delay_ms(300);
			strx = strstr((const char*)buf_uart2.buf, (const char*)"+CMS ERROR: 504");
		}
		printf("GPS启动命令已发送，等待GPS定位...\r\n");
		Clear_Buffer();
		
		do
		{
			delay_ms(10000);
			Uart2_SendStr("AT+QGPS?\r\n");
			delay_ms(300);
			strx = strstr((const char*)buf_uart2.buf, (const char*)"+QGPS: 1");
			if(strx)
			{
				printf("GPS定位已就绪\r\n");
				break;
			}
			printf("等待GPS定位中...\r\n");
			Clear_Buffer();
			retry_count++;
		}while(retry_count < 5);
		
		if(strx)
		{
			Clear_Buffer();
			Uart2_SendStr("AT+QGPSLOC=2\r\n");
			delay_ms(500);
			printf("定位数据:%s\r\n",buf_uart2.buf);
		}else{
			printf("GPS定位超时...\r\n");
		}
		Clear_Buffer();
	}
	IWDG_Feed();//喂狗
}

/*****************************************************
注册到ONENET
*****************************************************/
void CSTX_4G_RegALIYUNIOT(void)//平台注册
{
	int errcount = 0;
    Uart2_SendStr("AT+QMTDISC=0\r\n");
    delay_ms(300);
    Clear_Buffer();

    Uart2_SendStr("AT+QMTCLOSE=0\r\n");
    delay_ms(300);
    Clear_Buffer();
	
    delay_ms(300);
    Clear_Buffer();


    Uart2_SendStr("AT+QMTCFG=\"version\",0,4\r\n");//切换下版本
    delay_ms(300);  //等待300ms反馈OK
    strx=strstr((const char*)buf_uart2.buf,(const char*)"OK");//返OK
    while(strx==NULL)
    {
        strx=strstr((const char*)buf_uart2.buf,(const char*)"OK");//返OK
    }
    Clear_Buffer();

    Uart2_SendStr("AT+QMTOPEN=0,\"mqtts.heclouds.com\",1883\r\n");//登录到ONENET
    delay_ms(300);
    strx=strstr((const char*)buf_uart2.buf,(const char*)"+QMTOPEN: 0,0");//返+QMTOPEN: 0,0  +QMTOPEN: 0,0
    while(strx==NULL)
    {
		errcount++;
		delay_ms(30);
        strx=strstr((const char*)buf_uart2.buf,(const char*)"+QMTOPEN: 0,0");//返回OK
        if(errcount>10000)     //防止死循环
        {
            GPIO_SetBits(GPIOA,GPIO_Pin_12); 	
			delay_ms(1000);
			GPIO_ResetBits(GPIOA,GPIO_Pin_12); 	 
            delay_ms(300);
            NVIC_SystemReset();	//没有创建TCP SOCKET就重启系统等到服务器就绪
        }
    }
    Clear_Buffer();

    memset(ATSTR,0,BUFLEN);
    sprintf(ATSTR,"AT+QMTCONN=0,\"%s\",\"%s\",\"%s\"\r\n",DEVICENAME,PRODUCEKEY,DEVICESECRET); //+QMTCONN: 0,0,0
    printf("ATSTR = %s \r\n",ATSTR);
    Uart2_SendStr(ATSTR);//发送链接到阿里云
    delay_ms(300);
    strx=strstr((const char*)buf_uart2.buf,(const char*)"+QMTCONN: 0,0,0");//返+QMTCONN: 0,0,0
    while(strx==NULL)
    {
        strx=strstr((const char*)buf_uart2.buf,(const char*)"+QMTCONN: 0,0,0");//返+QMTCONN: 0,0,0
    }
    Clear_Buffer();
	printf("订阅成功\r\n");
}

/*********************************************************************************
** 函数名称 ： MakeOnenetPayload
** 功能描述 ： 生成包含温湿度、GPS坐标及运动参数的JSON载荷
** 输入参数 ： *Str - 输出缓冲区指针
**           Temp - 温度值（0-255）
**           Humi - 湿度值（0-255）
**           Latitude - 纬度（double类型）
**           Longitude - 经度（double类型）
**           Altitude - 海拔高度（米，double类型）
**			 Speed - 速度（米/秒，double类型）
**           Direction - 运动方向角度（0-360°，double类型）
** 返回值   ： JSON数据实际长度
*********************************************************************************/
u8 MakeOnenetPayload(u8 *Str, u8 Temp, u8 Humi, double Latitude, double Longitude, 
                    double Altitude, double Speed, double Direction) 
{
    char JsonBuf[256];  // 扩大缓冲区以容纳所有字段
    u16 JsonLen;
    
    // 生成包含温湿度、GPS及运动参数的JSON（OneNet物模型格式）
    sprintf(JsonBuf, 
            "{\"id\":\"123\",\"params\":{"
            "\"temp\":{\"value\":%d},"
            "\"humi\":{\"value\":%d},"
            "\"latitude\":{\"value\":%.6f},"
            "\"longitude\":{\"value\":%.6f},"
            "\"altitude\":{\"value\":%.2f},"
            "\"speed\":{\"value\":%.2f},"
			"\"direction\":{\"value\":%.1f}"
            
            "}}", 
            Temp, Humi, Latitude, Longitude, Altitude, Speed, Direction);
    
    JsonLen = strlen(JsonBuf);
    memcpy(Str, JsonBuf, JsonLen);
    return JsonLen;
}

/*********************************************************************************
** 函数名称 ： CSTX_4G_ALYIOTSenddata
** 功能描述 ： 发送温湿度、GPS数据及运动参数到OneNet平台
** 输入参数 ： temp - 温度值
**           humi - 湿度值
**           latitude - 纬度
**           longitude - 经度
**           altitude - 海拔高度（米）
**           direction - 运动方向角度（0-360°）
**           speed - 速度（米/秒）
** 关键流程 ： 1. 生成JSON载荷 → 2. MQTT发布 → 3. 验证结果
*********************************************************************************/
void CSTX_4G_ALYIOTSenddata(u8 temp, u8 humi, double latitude, double longitude, 
                           double altitude, double speed, double direction) 
{
    u8 errcount = 0;
    u8 SendBuf[256];
    u8 JsonLen = MakeOnenetPayload(SendBuf, temp, humi, latitude, longitude, 
                                  altitude, speed, direction);

    // 1. 订阅主题（可选）
    sprintf(ATSTR, "AT+QMTSUB=0,1,\"$sys/%s/%s/thing/property/set\",2\r\n", 
            PRODUCEKEY, DEVICENAME);
    Uart2_SendStr(ATSTR);
    delay_ms(300);
    Clear_Buffer();

    // 2. 发布数据（动态声明长度）
    sprintf(ATSTR, "AT+QMTPUBEX=0,0,0,0,\"$sys/%s/%s/thing/property/post\",%d\r\n", 
            PRODUCEKEY, DEVICENAME, JsonLen);
    Uart2_SendStr(ATSTR);
    delay_ms(300);

    // 3. 等待模块就绪（">"提示符）
    while (!strstr((char*)buf_uart2.buf, ">") && errcount++ < 100) {
        delay_ms(10);
    }
    Clear_Buffer();

    // 4. 发送JSON数据
    USART2_Send((char*)SendBuf, JsonLen);
    UART2_send_byte(0x1A);  // MQTT结束符
    delay_ms(500);  // 延长等待确保发送完成

    // 5. 验证发布结果
    errcount = 0;
    while (!strstr((char*)buf_uart2.buf, "+QMTPUBEX: 0,0,0") && errcount++ < 100) {
        delay_ms(10);
    }
    printf("数据已发送: 温度=%d, 湿度=%d, 纬度=%.6f, 经度=%.6f, 海拔高度=%.2fm, 速度=%.2fm/s, 方向=%.1f°\r\n", 
           temp, humi, latitude, longitude, altitude, speed, direction);
    Clear_Buffer();
}
