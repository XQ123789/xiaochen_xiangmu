/*********************************************************************************
** 函数名称 ： MakeOnenetPayload(char *Str,u8 Temp,u8 Humi)
** 函数功能 ： 将温度与湿度封装成Onenet有效载荷包
** 输    入	： *str有效载荷保存的目标字符串, 
** 输    入	： Temp 温度值
** 输    入	： Humi 湿度值
** 输    出	： 有效载荷字符串
** 返    回	： 有效载荷字符串长度
*********************************************************************************/
u8 MakeOnenetPayload(u8 *Str,u8 Temp,u8 Humi)//将温度与湿度封装成Onenet有效载荷包
{
	char Payload[] = "{\"datastreams\":[{\"id\":\"Temp\",\"datapoints\":[{\"value\":%d}]},{\"id\":\"Humi\",\"datapoints\":[{\"value\":%d}]}]}";//onenet有效载荷固定格式
	char JsonBuf[256];//临时数据缓存 
	unsigned short   StrLen;//有效载荷数据长度
	sprintf(JsonBuf,Payload,Temp,Humi);//合成数据有效载荷封装
	//printf("\r\n有效载荷(%d):[%s]\r\n",strlen(JsonBuf),(char *)JsonBuf);
	StrLen = strlen(JsonBuf)/sizeof(char);//计算有效载荷的数据长度
	//printf("\r\n有效载荷长度(%d)\r\n",StrLen);
	Str[0] = '\x01';//报文第一个字节的固定显示
	//printf("\r\nStr[0]=0X(%02X)\r\n",Str[0]);
	//报文第2个字节和第三个字节表示有效载荷的长度的固定显示
	Str[1] = (StrLen & 0XFF00) >>8;//高位
	//printf("\r\nStr[1]=0X(%02X)\r\n",Str[1]);
	Str[2] =  StrLen & 0XFF;//低位
	//printf("\r\nStr[2]=0X(%02X)\r\n",Str[2]);
	//拷贝有效载荷数据到Str
	memcpy(Str+3,JsonBuf,StrLen);//从第三个字节开始拷贝
	return (StrLen+3);//返回长度
}

/*********************************************************************************
** 函数名称 ： MakeOnenetPayload(char *Str, u8 Temp, u8 Humi)
** 函数功能 ： 将温度与湿度封装成OneNet MQTT协议要求的JSON格式
** 输    入 ： *Str - 有效载荷保存的目标字符串
**           Temp - 温度值
**           Humi - 湿度值
** 输    出 ： 有效载荷字符串
** 返    回 ： 有效载荷字符串长度
*********************************************************************************/
u8 MakeOnenetPayload(u8 *Str, u8 Temp, u8 Humi) 
{
    // 新版OneNet MQTT协议要求的JSON格式
    char Payload[] = "{\"id\":\"123\",\"params\":{\"humi\":{\"value\":%d},\"temp\":{\"value\":%d}}}";
    char JsonBuf[256]; // 临时数据缓存
    unsigned short StrLen; // 有效载荷数据长度

    // 1. 合成JSON数据
    sprintf(JsonBuf, Payload, Humi, Temp);
    StrLen = strlen(JsonBuf); // 计算有效载荷长度（不含头部的3字节）

    // 2. 构建MQTT报文头部（保持原框架不变）
    Str[0] = '\x01'; // 固定报文头
    Str[1] = (StrLen & 0xFF00) >> 8; // 长度高位
    Str[2] = StrLen & 0xFF; // 长度低位

    // 3. 拷贝JSON数据到输出缓冲区
    memcpy(Str + 3, JsonBuf, StrLen);
    
    return (StrLen + 3); // 返回总长度（头部3字节 + JSON数据）
}

/*********************************************************************************
** 函数名称 ： CSTX_4G_ALYIOTSenddata(u8 temp, u8 humi)
** 函数功能 ： 通过MQTT协议发送温湿度数据到OneNet平台
** 输    入 ： temp - 温度值
**           humi - 湿度值
*********************************************************************************/
u8   Send_Json[BUFLEN];//AT命令临时缓存
void CSTX_4G_ALYIOTSenddata(u8 temp, u8 humi) 
{
    u8 JsonLen;
    memset(Send_Json, 0, BUFLEN);
    
    // 1. 订阅主题（如果需要）
    Gui_DrawFont_GBK16(16,10,RED,WHITE, "SEND DATA...");
    Clear_Buffer();
    
    // 设置订阅主题（格式：$sys/{产品ID}/{设备名}/thing/property/set）
    sprintf(ATSTR, "AT+QMTSUB=0,1,\"$sys/%s/%s/thing/property/set\",2\r\n", PRODUCEKEY, DEVICENAME);
    Uart2_SendStr(ATSTR);
    delay_ms(300);
    
    // 等待订阅响应
    strx = strstr((const char*)buf_uart2.buf, ">");
    errcount = 0;
    while(strx == NULL) {
        errcount++;
        strx = strstr((const char*)buf_uart2.buf, ">");
        delay_ms(30);
        if(errcount > 100) break;
    }
    Clear_Buffer();
    
    // 2. 发布数据
    JsonLen = MakeOnenetPayload(Send_Json, temp, humi);
    
    // 设置发布命令（格式：$sys/{产品ID}/{设备名}/thing/property/post）
    memset(ATSTR, 0, BUFLEN);
    sprintf(ATSTR, "AT+QMTPUBEX=0,0,0,0,\"$sys/%s/%s/thing/property/post\",%d\r\n", 
            PRODUCEKEY, DEVICENAME, JsonLen);
    Uart2_SendStr(ATSTR);
    delay_ms(300);
    
    // 等待发布准备响应
    strx = strstr((const char*)buf_uart2.buf, ">");
    errcount = 0;
    while(strx == NULL) {
        errcount++;
        strx = strstr((const char*)buf_uart2.buf, ">");
        delay_ms(30);
        if(errcount > 100) break;
    }
    
    // 发送实际的JSON数据
    USART2_Send((char *)Send_Json, JsonLen);
    UART2_send_byte(0x1A);  // 发送结束符
    delay_ms(300);
    
    // 等待发布完成响应
    strx = strstr((const char*)buf_uart2.buf, "+QMTPUBEX: 0,0,0");
    errcount = 0;
    while(strx == NULL) {
        errcount++;
        strx = strstr((const char*)buf_uart2.buf, "+QMTPUBEX: 0,0,0");
        delay_ms(30);
        if(errcount > 100) break;
    }
    
    Clear_Buffer();
    Gui_DrawFont_GBK16(16,10,RED,WHITE, "SEND DATA OK");
}
